% "titlepage" gives the title its own page, as a thesis expects; without it the
% table of contents starts halfway down page 1, under the author's name.
\documentclass[12pt,a4paper,titlepage]{article}

\usepackage[T1]{fontenc}
\usepackage[margin=1in]{geometry}
\usepackage{booktabs}          % \toprule \midrule \bottomrule in the tables

\usepackage{microtype}
% hypertexnames=false: the titlepage option resets the page counter, so front
% matter repeats page numbers and hyperref would warn about duplicate anchors.
\usepackage[hidelinks,hypertexnames=false]{hyperref}
\usepackage{caption}
% Table captions sit ABOVE their tables (the generated body emits them first);
% this tells the caption package, so the caption-to-table gap is right.
\captionsetup[table]{position=top}

% Appendix B tabulates file names in \texttt{}. TeX will not break inside a
% typewriter word, so without these the longest row ran 133pt (~4.6cm) past the
% right margin. [htt] lets tt text hyphenate; \emergencystretch lets TeX loosen
% a line rather than overrun. Measured on this document, with and without these
% two lines: 48 overfull warnings across the build (worst 133.20pt) -> 0.
\usepackage[htt]{hyphenat}
\setlength{\emergencystretch}{3em}

\lstset{
  basicstyle=\ttfamily\footnotesize,
  breaklines=true,
  breakatwhitespace=true,
  columns=fullflexible,
  frame=single,
  rulecolor=\color{black!25},
  xleftmargin=1em,
  showstringspaces=false,
}
